28| function statusForPazarama(os: any): string {
29|   const n = Number(os);
30|   if (isNaN(n)) return String(os ?? '').toLowerCase();
31|   if (n <= 1) return 'pending';
32|   if (n <= 3) return 'processing';
33|   if (n <= 5) return 'shipped';
34|   if (n <= 7) return 'delivered';
35|   if (n <= 9) return 'cancelled';
36|   return 'returned';
